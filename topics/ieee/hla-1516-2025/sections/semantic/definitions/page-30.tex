\begin{hladefinition}[0pt]{def-used-for-sending}{used for sending}
\begin{hladefinitionlistdisplay}
\item Pertaining to a region and the specified interaction class designator: being an argument in the \HLAService{Send Interaction With Regions} service.
\item Pertaining to the default region: when the specified interaction class designator is an argument in the \HLAService{Send Interaction} service.
\HLADefinitionListNote[12.04pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\end{hladefinitionlistdisplay}
\end{hladefinition}

\begin{hladefinition}{def-used-for-subscription-of-a-class-attribute}{used for subscription of a class attribute}
\vspace*{1.05pt}
\begin{hladefinitionlistdisplay}
\item Pertaining to a region, an object class, and a class attribute: when the class attribute is a subscribed attribute of the object class with that region. See also: subscribed attributes of an object class with regions.
\HLADefinitionListNote[9.85pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\vspace*{3.84pt}
\item Pertaining to the default region: when the specified class attribute is a subscribed attribute of the specified class. See also: subscribed attributes of an object class with regions.
\HLADefinitionListNote[11.58pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\end{hladefinitionlistdisplay}
\end{hladefinition}

\begin{hladefinition}[12.10pt]{def-used-for-subscription-of-an-interaction-class}{used for subscription of an interaction class}
\vspace*{-1.02pt}
\begin{hladefinitionlistdisplay}
\item Pertaining to a region and an interaction class: when the interaction class is a subscribed interaction class with regions. \HLASeeAlso{\HLATermReference{subscribed interaction class}.}
\HLADefinitionListNote[11.78pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\vspace*{5.88pt}

\HLADefinitionListNote[11.84pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\end{hladefinitionlistdisplay}
\end{hladefinition}

\begin{hladefinition}[12.10pt]{def-used-for-update}{used for update}
\vspace*{0.18pt}
\begin{hladefinitionlistdisplaytightbottom}
\item Pertaining to a region that, along with the specified object instance and attribute designators, has been used as an argument in either the \HLAService{Register Object Instance With Regions} service or the \HLAService{Associate Regions For Updates} service: when the region has not subsequently been used along with the specified object instance designator as an argument in the \HLAService{Unassociate Regions For Updates} service, nor has the joined federate subsequently lost ownership of the specified instance attribute(s).
\HLADefinitionListNote[9.78pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\vspace*{3.90pt}
\item Pertaining to the default region: when the specified instance attribute(s) is not currently used for update with any other region.
\HLADefinitionListNote[7.80pt]{See 9.1 of IEEE Std 1516.1\HLAStandardMark-2025.}
\end{hladefinitionlistdisplaytightbottom}
\end{hladefinition}

\begin{hladefinition}{def-valid-federate-designator}{valid federate designator}
A federate designator which, during the federation execution, was returned by the \HLAService{Join Federation Execution} service to some joined federate. The federate does not have to remain a joined federate for its federate designator to remain a valid federate designator.
\end{hladefinition}

\begin{hladefinition}[7.98pt]{def-wall-clock-time}{wall-clock time}
Time as determined by a chronometer such as a wristwatch or a wall clock.
\end{hladefinition}

\begin{hladefinition}[7.98pt]{def-zero-time-interval}{zero time interval}
The distance between two logical time values that are equal. This time interval, when added to a logical time T1, results in a logical time T2 == T1.
\end{hladefinition}
