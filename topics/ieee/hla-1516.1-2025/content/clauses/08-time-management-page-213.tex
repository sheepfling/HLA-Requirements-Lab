  \HLAFixedBlock{90.000}{301.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{90.000}{325.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{89.999}{337.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{89.999}{349.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    execution may choose to change its class attribute's default order type. When changing a default order type,%
  }
  \HLAFixedBlock{89.999}{361.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    a class attribute shall be specified using a class handle for which the class attribute is available along with%
  }
  \HLAFixedBlock{89.999}{373.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{89.999}{397.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
