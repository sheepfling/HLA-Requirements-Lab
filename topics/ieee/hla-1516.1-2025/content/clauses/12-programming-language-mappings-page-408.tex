  \HLAFixedBlock{90.000}{69.646}{514.000}{\rmfamily\raggedright}{9.96}{11.16}{%
    \textbf{12.13.4.6 Rules for Sending Messages }%
  }
  \HLAFixedBlock{90.000}{93.785}{514.000}{\rmfamily\raggedright}{9.96}{11.16}{%
    The following rules apply for sending messages. %
  }
  \HLAFixedBlock{100.020}{117.785}{503.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    a)%
  }
  \HLAFixedBlock{121.970}{117.785}{482.030}{\rmfamily\raggedright}{9.96}{11.16}{%
    Establishing, maintaining and terminating a session%
  }
  \HLAFixedBlock{121.980}{129.785}{482.020}{\rmfamily\raggedright}{9.96}{11.16}{%
    1)%
  }
  \HLAFixedBlock{142.027}{129.785}{461.973}{\rmfamily\raggedright}{9.96}{11.16}{%
    No messages can be exchanged until a valid session has been established, with the exception of%
  }
  \HLAFixedBlock{142.020}{141.785}{461.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    Control messages for setting up such a session.%
  }
  \HLAFixedBlock{121.981}{153.784}{482.019}{\rmfamily\raggedright}{9.96}{11.16}{%
    2)%
  }
  \HLAFixedBlock{142.046}{153.784}{461.954}{\rmfamily\raggedright}{9.96}{11.16}{%
    Messages of the type CTRL\_NEW\_SESSION, CTRL\_TERMINATE\_SESSION and%
  }
  \HLAFixedBlock{142.020}{165.784}{461.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    CTRL\_HEARTBEAT shall only be sent by the federate.%
  }
  \HLAFixedBlock{121.981}{177.784}{482.019}{\rmfamily\raggedright}{9.96}{11.16}{%
    3)%
  }
  \HLAFixedBlock{142.029}{177.784}{461.971}{\rmfamily\raggedright}{9.96}{11.16}{%
    Messages of the type CTRL\_NEW\_SESSION\_STATUS, CTRL\_HEARTBEAT\_RESPONSE%
  }
  \HLAFixedBlock{142.021}{189.784}{461.979}{\rmfamily\raggedright}{9.96}{11.16}{%
    and CTRL\_SESSION\_TERMINATED shall only be sent by the RTI.%
  }
  \HLAFixedBlock{121.982}{201.784}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    4)%
  }
  \HLAFixedBlock{142.035}{201.784}{461.965}{\rmfamily\raggedright}{9.96}{11.16}{%
    For every message of the type CTRL\_NEW\_SESSION received by the RTI, it shall send a cor-%
  }
  \HLAFixedBlock{142.022}{213.783}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    responding CTRL\_NEW\_SESSION\_STATUS message.%
  }
  \HLAFixedBlock{121.982}{225.783}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    5)%
  }
  \HLAFixedBlock{142.035}{225.783}{461.965}{\rmfamily\raggedright}{9.96}{11.16}{%
    For every message of the type CTRL\_HEARTBEAT received by the RTI, it shall send a corre-%
  }
  \HLAFixedBlock{142.022}{237.783}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    sponding CTRL\_HEARTBEAT\_RESPONSE message.%
  }
  \HLAFixedBlock{121.982}{249.783}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    6)%
  }
  \HLAFixedBlock{142.041}{249.783}{461.959}{\rmfamily\raggedright}{9.96}{11.16}{%
    For every message of the type CTRL\_TERMINATE SESSION received by the RTI, it shall%
  }
  \HLAFixedBlock{142.023}{261.783}{461.977}{\rmfamily\raggedright}{9.96}{11.16}{%
    send a corresponding CTRL\_SESSION\_TERMINATED message.%
  }
  \HLAFixedBlock{121.983}{273.783}{482.017}{\rmfamily\raggedright}{9.96}{11.16}{%
    7)%
  }
  \HLAFixedBlock{142.038}{273.783}{461.962}{\rmfamily\raggedright}{9.96}{11.16}{%
    When the federate receives a CTRL\_SESSION\_TERMINATED message, it shall close the%
  }
  \HLAFixedBlock{142.023}{285.782}{461.977}{\rmfamily\raggedright}{9.96}{11.16}{%
    connection.%
  }
  \HLAFixedBlock{121.983}{297.782}{482.017}{\rmfamily\raggedright}{9.96}{11.16}{%
    8)%
  }
  \HLAFixedBlock{142.036}{297.782}{461.964}{\rmfamily\raggedright}{9.96}{11.16}{%
    With the exception of the NEW\_SESSION message, if the RTI or federate receive a message%
  }
  \HLAFixedBlock{142.023}{309.782}{461.977}{\rmfamily\raggedright}{9.96}{11.16}{%
    with an invalid Session ID or sequence number, it shall close the socket and abort the session. If%
  }
  \HLAFixedBlock{142.022}{321.782}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    the RTI closes the socket, the federate may then try to resume the session, but the recovery shall%
  }
  \HLAFixedBlock{142.022}{333.782}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    fail.%
  }
  \HLAFixedBlock{100.022}{345.781}{503.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    b)%
  }
  \HLAFixedBlock{121.949}{345.781}{482.051}{\rmfamily\raggedright}{9.96}{11.16}{%
    HLA calls and callbacks%
  }
  \HLAFixedBlock{121.982}{357.781}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    1)%
  }
  \HLAFixedBlock{142.033}{357.781}{461.967}{\rmfamily\raggedright}{9.96}{11.16}{%
    Messages of type HLA\_CALL\_REQUEST and HLA\_CALLBACK\_RESPONSE SHALL%
  }
  \HLAFixedBlock{142.021}{369.781}{461.979}{\rmfamily\raggedright}{9.96}{11.16}{%
    only be sent by the federate.%
  }
  \HLAFixedBlock{121.982}{381.781}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    2)%
  }
  \HLAFixedBlock{142.034}{381.781}{461.966}{\rmfamily\raggedright}{9.96}{11.16}{%
    Messages of the type HLA\_CALL\_RESPONSE and HLA\_CALLBACK\_REQUEST shall%
  }
  \HLAFixedBlock{142.021}{393.781}{461.979}{\rmfamily\raggedright}{9.96}{11.16}{%
    only be sent by the RTI.%
  }
  \HLAFixedBlock{121.982}{405.780}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    3)%
  }
  \HLAFixedBlock{142.038}{405.780}{461.962}{\rmfamily\raggedright}{9.96}{11.16}{%
    For every message of the type HLA\_CALL\_REQUEST received by the RTI, it shall send a%
  }
  \HLAFixedBlock{142.022}{417.780}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    corresponding HLA\_CALL\_RESPONSE including the sequence number of the  HLA\_-%
  }
  \HLAFixedBlock{142.022}{429.780}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    CALL\_REQUEST.%
  }
  \HLAFixedBlock{121.982}{441.780}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    4)%
  }
  \HLAFixedBlock{142.026}{441.780}{461.974}{\rmfamily\raggedright}{9.96}{11.16}{%
    The federate may send additional HLA\_CALL\_REQUESTs before an HLA\_CALL\_RE-%
  }
  \HLAFixedBlock{142.023}{453.780}{461.977}{\rmfamily\raggedright}{9.96}{11.16}{%
    SPONSE for an already sent messages is received.%
  }
  \HLAFixedBlock{121.983}{465.780}{482.017}{\rmfamily\raggedright}{9.96}{11.16}{%
    5)%
  }
  \HLAFixedBlock{142.039}{465.780}{461.961}{\rmfamily\raggedright}{9.96}{11.16}{%
    For every message of the type HLA\_CALLBACK\_REQUEST received by the federate, it shall%
  }
  \HLAFixedBlock{142.023}{477.779}{461.977}{\rmfamily\raggedright}{9.96}{11.16}{%
    send a corresponding HLA Callback Response, including the sequence number of the HLA\_-%
  }
  \HLAFixedBlock{142.022}{489.779}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    CALLBACK\_REQUEST.%
  }
  \HLAFixedBlock{121.983}{501.779}{482.017}{\rmfamily\raggedright}{9.96}{11.16}{%
    6)%
  }
  \HLAFixedBlock{142.033}{501.779}{461.967}{\rmfamily\raggedright}{9.96}{11.16}{%
    The RTI may send additional HLA\_CALLBACK\_REQUESTs before an HLA\_CALL-%
  }
  \HLAFixedBlock{142.022}{513.779}{461.978}{\rmfamily\raggedright}{9.96}{11.16}{%
    BACK\_RESPONSE for an already sent messages is received.%
  }
  \HLAFixedBlock{121.982}{525.779}{482.018}{\rmfamily\raggedright}{9.96}{11.16}{%
    7)%
  }
  \HLAFixedBlock{142.033}{525.779}{461.967}{\rmfamily\raggedright}{9.96}{11.16}{%
    The RTI shall process one received HLA\_CALLBACK\_REQUEST at a time including issu-%
  }
  \HLAFixedBlock{142.021}{537.778}{461.979}{\rmfamily\raggedright}{9.96}{11.16}{%
    ance of a response. The federate shall process one received HLA\_CALLBACK\_REQUEST at%
  }
  \HLAFixedBlock{142.021}{549.778}{461.979}{\rmfamily\raggedright}{9.96}{11.16}{%
    a time including issuance of a response.%
  }
  \HLAFixedBlock{100.021}{561.778}{503.979}{\rmfamily\raggedright}{9.96}{11.16}{%
    c)%
  }
  \HLAFixedBlock{121.981}{561.778}{482.019}{\rmfamily\raggedright}{9.96}{11.16}{%
    Reconnection%
  }
  \HLAFixedBlock{121.981}{573.778}{482.019}{\rmfamily\raggedright}{9.96}{11.16}{%
    1)%
  }
  \HLAFixedBlock{142.035}{573.778}{461.965}{\rmfamily\raggedright}{9.96}{11.16}{%
    Messages of the type CTRL\_RESUME\_REQUEST shall only be sent by the federate.%
  }
  \HLAFixedBlock{121.981}{585.778}{482.019}{\rmfamily\raggedright}{9.96}{11.16}{%
    2)%
  }
  \HLAFixedBlock{142.037}{585.778}{461.963}{\rmfamily\raggedright}{9.96}{11.16}{%
    Messages of the type CTRL\_RESUME\_STATUS shall only be sent by the RTI.%
  }
  \HLAFixedBlock{121.981}{597.777}{482.019}{\rmfamily\raggedright}{9.96}{11.16}{%
    3)%
  }
  \HLAFixedBlock{142.036}{597.777}{461.964}{\rmfamily\raggedright}{9.96}{11.16}{%
    For every message of the type CTRL\_RESUME\_REQUEST received by the RTI, it shall send%
  }
  \HLAFixedBlock{142.020}{609.777}{461.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    a corresponding RESUME\_STATUS Control message indicating whether the reconnection can%
  }
  \HLAFixedBlock{142.020}{621.777}{461.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    take place.%
  }
  \HLAFixedBlock{90.000}{645.777}{514.000}{\rmfamily\raggedright}{9.96}{11.16}{%

    operation over the transport layer.%
  }
