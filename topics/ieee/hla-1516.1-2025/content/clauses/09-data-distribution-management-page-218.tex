  \HLAFixedBlock{142.024}{159.784}{379.976}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{142.021}{186.784}{379.979}{\rmfamily\raggedright}{9.96}{12.00}{%

    dimension for that region.%
  }
  \HLAFixedBlock{121.982}{213.783}{400.018}{\rmfamily\raggedright}{9.96}{12.00}{%
    3)%
  }
  \HLAFixedBlock{142.021}{213.783}{379.979}{\rmfamily\raggedright}{9.96}{12.00}{%
    Invoke \emph{Commit Region Modifications} (see 9.3) to inform the RTI about the changes to the%
  }
  \HLAFixedBlock{142.022}{225.783}{379.978}{\rmfamily\raggedright}{9.96}{12.00}{%

  }
  \HLAFixedBlock{142.022}{237.783}{379.978}{\rmfamily\raggedright}{9.96}{12.00}{%
    invocations.%
  }
  \HLAFixedBlock{121.983}{252.783}{400.017}{\rmfamily\raggedright}{9.96}{12.00}{%
    A region template is created when a federate invokes the \emph{Create Region} service. The region designa-%

  \HLAFixedBlock{121.983}{264.783}{400.017}{\rmfamily\raggedright}{9.96}{12.00}{%
    tor argument that is returned as a result of the \emph{Create Region} service is a designator of a region tem-%
  }
  \HLAFixedBlock{121.984}{276.782}{400.016}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{121.984}{300.782}{400.016}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{121.985}{324.782}{400.015}{\rmfamily\raggedright}{9.96}{12.00}{%
