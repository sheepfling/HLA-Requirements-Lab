  }
  \HLAFixedBlock{90.000}{94.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Unassociate Regions For Updates }service shall remove the association between the regions and the%
  }
  \HLAFixedBlock{89.999}{106.806}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    specified instance attributes. If one or more of the specified regions are in the set of associations, then these%
  }
  \HLAFixedBlock{89.999}{118.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    regions shall be unassociated from the specified instance attributes; if any of the specified regions are not in%
  }
  \HLAFixedBlock{89.999}{130.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    the set of associations of the specified instance attributes, they shall be ignored.%
  }
  \HLAFixedBlock{90.000}{154.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    Each of the instance attributes that are unassociated by the invocation of this service shall be associated with%
  }
  \HLAFixedBlock{90.000}{166.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    the default region if and only if they are not associated with any other update region.%
  }
  \HLAFixedBlock{90.000}{190.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    If the region set paired with an instance attribute is empty, no associations shall be removed for that instance%
  }
  \HLAFixedBlock{90.000}{202.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    attribute.%
  }
