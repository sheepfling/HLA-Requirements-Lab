    of a given joined federate, are%
  }

  \HLAFixedAlphaList{100.02}{153.78}{421.98}{6}{%
    \item Unowned instance attributes of object instances that have a known class at the joined federate equal\linebreak
      to the object class of the class attribute and that have the same attribute designator as the class\linebreak
      attribute, or
    \item Instance attributes owned by the joined federate that belong to object instances that have a known\linebreak
      class at the owning federate equal to the object class of the class attribute and that have the same\linebreak
      attribute designator as the class attribute.
  }

  \HLADefinitionAt{243.78}{%
    \HLADefinitionAnchor{current-federation-object-model}\HLADefinitionTerm{current federation object model (FOM)}: The union of the FOM modules and one Management Object\linebreak
    Model (MOM) and Initialization Module (MIM) that have been specified in the creation of the federation\linebreak
    execution or by any federate that has joined the federation execution. The sum operation is carried out\linebreak
    according to the rules as prescribed in IEEE Std 1516.2\texttrademark-2025. When all FOM modules have been\linebreak
    provided, the current FOM will be equal to the FOM; and before this step has happened, the current FOM\linebreak
    will be a true subset of the FOM.%
  }

  \HLADefinitionAt{327.78}{%
    \HLADefinitionAnchor{current-federation-object-model-document-data}\HLADefinitionTerm{current federation object model (FOM) Document Data (FDD)}: The data and information used to\linebreak
    configure the federation execution that is the union of the FOM modules and one Management Object\linebreak
    Model (MOM) and Initialization Module (MIM) that have been specified in the creation of the federation\linebreak
    execution or by any federate that has joined the federation execution. The sum operation is carried out\linebreak
    according to the rules as prescribed in IEEE Std 1516.2\texttrademark-2025. When all FOM modules have been\linebreak
    provided, the current FDD will be equal to the FDD; and before this step has happened, the current FDD will\linebreak
    be a true subset of the FDD.%
  }

  }

  \HLADefinitionAt{579.78}{%
    \HLADefinitionAnchor{default-range}\HLADefinitionTerm{default range}: A range lower bound and a range upper bound, defined in the federation object model\linebreak
    (FOM) Document Data (FDD) and specified in terms of [0, the dimension's upper bound), for a dimension.%
  }

  \HLADefinitionAt{615.78}{%
    \HLADefinitionAnchor{default-region}\HLADefinitionTerm{default region}: A multidimensional region provided by the runtime infrastructure (RTI) that is composed of\linebreak
    one range for each dimension found in the federation object model (FOM) Document Data (FDD). The\linebreak
    bounds of each of these ranges are [0, the range's dimension's upper bound). There is no way for a federate\linebreak
    to refer to the default region.%
  }
