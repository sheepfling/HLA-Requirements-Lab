  \HLAServiceHeadingAt{303.383}{%
    4.24 \emph{Abort Federation Save} service%
  }
  \HLAFixedBlock{90.000}{328.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Abort Federation Save} service requests the RTI to abort the federation save that is underway. %
  }
  \HLAFixedBlock{90.001}{352.805}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    The service takes no arguments and affects the current save. This service may be invoked only while the%
  }
  \HLAFixedBlock{90.001}{364.805}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    federate is in the Federate Save in Progress state. Upon return of the service invocation, a request to abort the%
  }
  \HLAFixedBlock{90.001}{376.805}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    current save has successfully been made.%
  }
  \HLAFixedBlock{90.001}{400.804}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    The RTI shall abort the current save by invoking the \emph{Federation Saved \textdagger{} }service on all federates that are in%
  }
  \HLAFixedBlock{90.001}{412.804}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    the Federate Save in Progress state, regardless of each individual federate’s save substate. Joined federates%
  }
  \HLAFixedBlock{90.001}{424.804}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    not in the Federation Save in Progress state may not receive the \emph{Initiate Federation Saved \textdagger{}} callback for this%
  }
  \HLAFixedBlock{90.001}{436.804}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    federation save. The save-success indicator for the \emph{Federation Saved \textdagger{}} service shall indicate failure%
  }
  \HLAFixedBlock{90.001}{448.804}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    (federate save aborted) if not all federates have successfully invoked the \emph{Federate Save Complete} service%
  }
  \HLAFixedBlock{90.001}{460.804}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    with a save-success indicator that indicated success. Since a federate remains in the Federate Save in%
  }
  \HLAFixedBlock{90.001}{472.803}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    Progress state after it successfully invokes the \emph{Federate Save Complete} service and before the \emph{Federation}%
  }
  \HLAFixedBlock{90.001}{484.803}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Saved \textdagger{}} service is invoked on it by the RTI, it is possible that all other federates may have invoked the%
  }
  \HLAFixedBlock{90.001}{496.803}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Federate Save Complete} service with a save-success indicator of indicating success. In this case, it is%
  }
  \HLAFixedBlock{90.001}{508.803}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    possible that the \emph{Federation Saved \textdagger{} }service is invoked at each federate with a save-success indicator%

  \HLAFixedBlock{90.001}{520.803}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    indicating success after a invocation of the \emph{Abort Federation Save} service.%
