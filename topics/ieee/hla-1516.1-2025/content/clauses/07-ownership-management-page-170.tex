  \HLAServiceHeadingAt{69.383}{%
    \textbf{7.15 }\textbf{\emph{Cancel Attribute Ownership Acquisition}}\textbf{ service}%
  }
  \HLAFixedBlock{90.000}{94.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Cancel Attribute Ownership Acquisition} service shall notify the RTI that the joined federate no longer%
  }
  \HLAFixedBlock{90.000}{106.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    wants to acquire ownership of the specified instance attributes. This service shall always receive one of three%
  }
  \HLAFixedBlock{90.000}{118.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    replies from the RTI. The first form of reply, \emph{Confirm Attribute Ownership Acquisition Cancellation}, shall%
  }
  \HLAFixedBlock{90.001}{130.805}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    indicate that the request to acquire ownership of the specified instance attributes has been successfully%
  }
  \HLAFixedBlock{90.001}{142.805}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    canceled. The second form of reply, \emph{Attribute Ownership Acquisition Notification \textdagger{}}, shall indicate that the%
  }
  \HLAFixedBlock{90.001}{154.805}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    request to acquire ownership of the specified instance attributes was not canceled before the joined federate%
  }
  \HLAFixedBlock{90.002}{166.805}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    acquired ownership of the instance attributes. The third form of reply, \emph{Attribute Ownership Unavailable \textdagger{}},%
  }
  \HLAFixedBlock{90.002}{178.804}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    shall indicate that the request to acquire ownership of the specified instance attributes was not canceled%
  }
  \HLAFixedBlock{90.002}{190.804}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    before the joined federate was denied ownership of the instance attributes. The joined federate may receive%
  }
  \HLAFixedBlock{90.002}{202.804}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    multiple forms of reply in response to a single \emph{Cancel Attribute Ownership Acquisition} service invocation%
  }
  \HLAFixedBlock{90.002}{214.804}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    since the cancellation may succeed for some of the supplied instance attributes and fail for others. This%
  }
  \HLAFixedBlock{90.002}{226.804}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    service shall be used only to cancel requests to acquire ownership of instance attributes that were made via%
  }
  \HLAFixedBlock{90.002}{238.804}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    the \emph{Attribute Ownership Acquisition} service. Requests made via the \emph{Attribute Ownership Acquisition If}%
  }
  \HLAFixedBlock{90.002}{250.803}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Available} service shall not be explicitly canceled. They may, however, be overridden by an invocation of the%
  }
  \HLAFixedBlock{90.002}{262.803}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Attribute Ownership Acquisition} service.%
  }
