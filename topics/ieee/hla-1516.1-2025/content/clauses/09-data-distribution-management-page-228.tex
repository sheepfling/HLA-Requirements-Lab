    \textbf{9.5 }\textbf{\emph{Register Object Instance With Regions}}\textbf{ service}%
  }
  \HLAFixedBlock{90.000}{556.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Register Object Instance With Regions} service shall create a unique object instance designator and link%
  }
  \HLAFixedBlock{90.000}{568.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    it with an object instance of the supplied object class. All instance attributes of the object instance for which%
  }
  \HLAFixedBlock{90.000}{580.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    the corresponding class attributes are currently published by the registering joined federate shall be set as%
  }
  \HLAFixedBlock{90.000}{592.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    owned by the registering joined federate.%
  }
  \HLAFixedBlock{90.000}{616.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    This service shall be used to create an object instance and simultaneously associate update regions with%
  }
  \HLAFixedBlock{90.000}{628.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    instance attributes of that object instance. This service shall be an atomic operation that can be used in place%
  }
  \HLAFixedBlock{90.000}{640.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    of the \emph{Register Object Instance} service followed by the \emph{Associate Regions For Updates} service when the%
  }
  \HLAFixedBlock{90.000}{652.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    atomicity is required. The instance attributes whose corresponding class attributes are currently published%
  }
  \HLAFixedBlock{90.000}{664.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    and that have available dimensions, but are not supplied in the service invocation, shall be associated with%
  }
  \HLAFixedBlock{89.999}{676.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    the default region.%
  }
