  \HLAFixedBlock{90.000}{165.645}{514.000}{\rmfamily\raggedright}{9.96}{11.16}{%
    \textbf{12.13.5.1 Special consideration for TLS 1.3 }%
  }
  \HLAFixedBlock{90.000}{189.784}{514.000}{\rmfamily\raggedright}{9.96}{11.16}{%
    An RTI and a federate may optionally support TLS 1.3 transportation. When using TLS 1.3 the following%
  }
  \HLAFixedBlock{90.000}{201.784}{514.000}{\rmfamily\raggedright}{9.96}{11.16}{%
    modes shall be supported by the RTI and the Federate:%
  }
  \HLAFixedBlock{100.020}{225.783}{503.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    a)%
  }
  \HLAFixedBlock{121.975}{225.783}{482.025}{\rmfamily\raggedright}{9.96}{11.16}{%
    No certificate validation, i.e. encryption only.%
  }
  \HLAFixedBlock{100.020}{237.783}{503.980}{\rmfamily\raggedright}{9.96}{11.16}{%
    b)%
  }
  \HLAFixedBlock{121.942}{237.783}{482.058}{\rmfamily\raggedright}{9.96}{11.16}{%
    Server (RTI) certificate validation. In this case a server certificate and a corresponding private key%
  }
  \HLAFixedBlock{121.980}{249.783}{482.020}{\rmfamily\raggedright}{9.96}{11.16}{%
    shall be installed at the RTI. The federate shall verify the validity of the server certificate and the%
  }
  \HLAFixedBlock{121.980}{261.783}{482.020}{\rmfamily\raggedright}{9.96}{11.16}{%
    identity of the RTI using a trusted certificate. %
  }
