  \HLADefinitionAt{278.76}{%
    \HLADefinitionAnchor{simulation-object-model}\HLADefinitionTerm{simulation object model (SOM)}: A specification of the types of information that an individual federate\linebreak
    could provide to High Level Architecture (HLA) federations as well as the information that an individual\linebreak
    federate can receive from other federates in HLA federations. The SOM is specified using one or more SOM\linebreak
    modules. The standard format in which SOMs are expressed facilitates determination of the suitability of\linebreak
    federates for participation in a federation.%
  }

  \HLADefinitionAt{350.76}{%
    \HLADefinitionAnchor{simulation-object-model-module}\HLADefinitionTerm{simulation object model (SOM) module}: A subset of the SOM that contains some or all object model\linebreak
    template (OMT) tables. SOM modules and one optional Management Object Model (MOM) and\linebreak
    Initialization Module (MIM) are used to specify the SOM of a federate. A SOM module contains complete\linebreak
    or scaffolding definitions for all object classes and interaction classes that are superclasses of object classes\linebreak
    and interaction classes in the same SOM module.%
  }
