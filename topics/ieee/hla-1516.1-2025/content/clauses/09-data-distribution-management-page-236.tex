  \HLAFixedBlock{90.000}{496.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  }
  \HLAFixedBlock{90.000}{508.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    operations shall behave analogously to the \emph{Subscribe Interaction Class }service as described in 5.10. This%
  }
  \HLAFixedBlock{90.000}{520.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    service shall provide additional functionality in that the overlap of the region set used for subscription of the%
  }
  \HLAFixedBlock{90.000}{532.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    interaction and the region set used for sending the interaction affects the subsequent RTI operations, as%
  }
  \HLAFixedBlock{90.000}{544.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    described in 9.1.5.%
  }
  \HLAFixedBlock{90.000}{568.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    This service shall add the specified regions to the set of regions associated with the specified interaction%
  }
  \HLAFixedBlock{90.000}{580.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    class subscription. No changes shall be made to the subscription set if the specified regions are already in the%
  }
  \HLAFixedBlock{90.000}{592.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    set of subscriptions of the specified interaction class.%
  }
  \HLAFixedBlock{90.000}{616.803}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    Invocations of the \emph{Subscribe Interaction Class With Regions }service shall have no affect on any interaction%

  \HLAFixedBlock{90.000}{628.803}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{90.000}{640.803}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
