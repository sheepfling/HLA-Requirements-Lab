  \HLAFixedBlock{90.000}{144.786}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  }
  \HLAFixedBlock{90.000}{155.766}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{90.000}{166.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  }
  \HLAFixedBlock{90.000}{177.785}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    owned by the registering joined federate.%
  }
  \HLAFixedBlock{90.000}{198.785}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    An object instance handle shall be uniform throughout the federation execution; i.e., with respect to a%
  }
  \HLAFixedBlock{90.000}{209.765}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    particular object instance, the handle provided to the joined federate that registers the object instance shall%
  }
  \HLAFixedBlock{90.000}{220.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{90.001}{231.784}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%

  }
  \HLAFixedBlock{90.001}{242.764}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{90.002}{264.784}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    between joined federates that are using different language APIs. The handles may be communicated%

  \HLAFixedBlock{90.002}{275.764}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%

  \HLAFixedBlock{90.002}{307.803}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%

    name, and that name shall be coadunated with the object instance.%
