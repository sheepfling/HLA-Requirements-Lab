  \HLAServiceHeadingAt{69.383}{%
    4.20 \emph{Initiate Federate Save \textdagger{}} service%
  }
  \HLAFixedBlock{90.000}{94.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Initiate Federate Save \textdagger{}} service shall instruct the joined federate to save state. The joined federate%
  }
  \HLAFixedBlock{90.000}{106.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    should save as soon as possible after the invocation of the \emph{Initiate Federate Save \textdagger{}} service. The label%
  }
  \HLAFixedBlock{90.000}{118.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    provided to the RTI when the save was requested, via the \emph{Request Federation Save} service, shall be supplied%
  }
  \HLAFixedBlock{89.999}{130.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    to the joined federate. The joined federate shall use this label, the name of the federation execution, its%
  }
  \HLAFixedBlock{89.999}{142.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    joined federate designator, and its federate type, which it supplied when it invoked the \emph{Join Federation}%
  }
  \HLAFixedBlock{89.999}{154.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Execution} service, to distinguish the saved state information. If a joined federate is not time constrained, it%
  }
  \HLAFixedBlock{89.999}{166.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    shall expect to receive an \emph{Initiate Federate Save \textdagger{}} service invocation at any point. If a joined federate is%
  }
  \HLAFixedBlock{90.000}{178.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    time-constrained, it shall expect to receive an \emph{Initiate Federate Save \textdagger{}} service invocation only when it is in%
  }
  \HLAFixedBlock{90.000}{190.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    the Time Advancing state. The joined federate shall stop providing new information to the federation%
  }
  \HLAFixedBlock{90.000}{202.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    immediately after receiving the \emph{Initiate Federate Save \textdagger{}} service invocation. The joined federate may resume%
  }
  \HLAFixedBlock{90.000}{214.804}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    providing new information to the federation only after receiving the \emph{Federation Saved \textdagger{} }service invocation.%
  }
  \HLAFixedBlock{89.999}{238.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    If the corresponding \emph{Request Federation Save} service invocation had a timestamp argument, that timestamp%
  }
  \HLAFixedBlock{89.999}{250.803}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    shall be provided to the resulting \emph{Initiate Federate Save\textdagger{} }service invocation.%
  }
