  \HLAServiceComponentHeadingAt{279.643}{%
    \textbf{7.1.5 Sets of attribute designators}%
  }
  \HLAFixedBlock{90.001}{303.782}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    Although many of the ownership management services take a set of instance attributes as an argument, the%
  }
  \HLAFixedBlock{90.001}{315.782}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    RTI treats ownership management operations on a per-instance-attribute basis. The fact that some ownership%
  }
  \HLAFixedBlock{90.001}{327.782}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    management service invocations take sets of instance attributes as an argument is a feature provided to%
  }
  \HLAFixedBlock{90.001}{339.781}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    joined federate designers for convenience. A single request with an instance attribute set as an argument can%
  }
  \HLAFixedBlock{90.001}{351.781}{431.999}{\rmfamily\raggedright}{9.96}{12.00}{%
    result in multiple responses pertaining to disjoint subsets of those instance attributes. For example, a single%
  }
  \HLAFixedBlock{90.002}{363.781}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Negotiated Attribute Ownership Divestiture} service invocation that has a set of instance attributes as an%
  }
  \HLAFixedBlock{90.002}{375.781}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    argument could result in multiple \emph{Request Divestiture Confirmation \textdagger{}} service invocations. If one instance%
  }
  \HLAFixedBlock{90.002}{387.781}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    attribute in the set of instance attributes provided as an argument to an ownership management service%
  }
  \HLAFixedBlock{90.002}{399.780}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    invocation violates the preconditions of the service, an exception shall be generated, and the entire service%
  }
  \HLAFixedBlock{90.002}{411.780}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    invocation shall fail. The RTI shall treat all joined federate requests to divest or acquire attributes separately.%
  }
  \HLAFixedBlock{90.002}{423.780}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    Responses to or requests of joined federates shall not contain information from separate joined federate%
  }
  \HLAFixedBlock{90.002}{435.780}{431.998}{\rmfamily\raggedright}{9.96}{12.00}{%
    requests.%
  }
  \HLAServiceHeadingAt{465.383}{%
    \textbf{7.2 }\textbf{\emph{Unconditional Attribute Ownership Divestiture}}\textbf{ service}%
  }
  \HLAFixedBlock{90.000}{490.806}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Unconditional Attribute Ownership Divestiture} service shall notify the RTI that the joined federate no%
  }
  \HLAFixedBlock{90.000}{502.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    longer wants to own the specified instance attributes of the specified object instance. This service shall%
  }
  \HLAFixedBlock{90.000}{514.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    immediately relieve the divesting joined federate of the ownership and cause the instance attribute(s) to go%
  }
  \HLAFixedBlock{90.000}{526.805}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    (possibly temporarily) into the unowned state, without regard to the existence of an accepting joined%
  }
  \HLAFixedBlock{89.999}{538.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    federate.%
  }
  \HLAFixedBlock{89.999}{562.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    For each instance attribute that becomes unowned as a result of invocation of this service, if no joined%
  }
  \HLAFixedBlock{89.999}{574.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    federates are in either the Acquiring state or the Willing to Acquire state with respect to the specified%
  }
  \HLAFixedBlock{89.999}{586.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    instance attribute, the RTI shall try to identify other joined federates that are willing to own the instance%
  }
  \HLAFixedBlock{89.999}{598.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    attribute. If any joined federate is in either the Acquiring state or the Willing to Acquire state with respect to%
  }
  \HLAFixedBlock{89.999}{610.804}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    the specified instance attribute, the RTI may try to identify other joined federates that are willing to own the%
  }
  \HLAFixedBlock{89.999}{622.803}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    instance attribute. The mechanism that the RTI shall use to try to identify other joined federates that are%
  }
  \HLAFixedBlock{89.999}{634.803}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    willing to own the instance attribute is invocation of the \emph{Request Attribute Ownership Assumption \textdagger{}} service%
  }
  \HLAFixedBlock{89.999}{646.803}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    at other joined federates (joined federates other than the divesting joined federate) that are both eligible to%
  }
  \HLAFixedBlock{89.999}{658.803}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    own the instance attribute and not in either the Acquiring state or the Willing to Acquire state with respect to%
  }
  \HLAFixedBlock{89.999}{670.803}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    the specified instance attribute. As long as the instance attribute remains unowned, the RTI shall try to%
  }
  \HLAFixedBlock{89.999}{682.802}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    identify joined federates (other than the divesting joined federate) that are willing to own the instance%
  }
  \HLAFixedBlock{89.998}{694.802}{432.002}{\rmfamily\raggedright}{9.96}{12.00}{%
    attribute; but once the instance attribute becomes owned, the RTI shall not invoke the \emph{Request Attribute}%
  }
  \HLAFixedBlock{89.998}{706.802}{432.002}{\rmfamily\raggedright}{9.96}{12.00}{%
    \emph{Ownership Assumption \textdagger{}} service at any additional joined federates.%
  }
