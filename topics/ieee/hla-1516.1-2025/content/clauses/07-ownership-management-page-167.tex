  }
  \HLAFixedBlock{89.998}{130.805}{432.002}{\rmfamily\raggedright}{9.96}{12.00}{%
    posed as a result of the invocation of \emph{Request Attribute Ownership Release \textdagger{}}, or it may use this service for%
  }
  \HLAFixedBlock{89.998}{142.805}{432.002}{\rmfamily\raggedright}{9.96}{12.00}{%
    any other reason. Invocation of this service for attributes that have no pending acquisition has no effect.%
  }
  \HLAFixedBlock{89.998}{154.805}{432.002}{\rmfamily\raggedright}{9.96}{12.00}{%
    Invocation of the \emph{Attribute Ownership Release Denied} service shall cause the RTI to terminate any pending%

  \HLAFixedBlock{89.999}{166.805}{432.001}{\rmfamily\raggedright}{9.96}{12.00}{%
    acquisitions and invoke \emph{Attribute Ownership Unavailable \textdagger{}} at the acquiring joined federates.%
