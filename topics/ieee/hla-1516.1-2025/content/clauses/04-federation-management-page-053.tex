  \HLAServiceHeadingAt{219.383}{%
    4.5 \emph{Create Federation Execution} service%
  }
  \HLAFixedBlock{90.000}{244.806}{433.270}{\rmfamily}{9.96}{12.00}{%
    The \emph{Create Federation Execution }service shall create a new federation execution and add it to the set of\linebreak%
    supported federation executions. Each federation execution created by this service shall be independent of\linebreak%
    all other federation executions, and there shall be no intercommunication within the RTI between federation\linebreak%
    executions. If the RTI is configured to require authorization, the RTI shall require that the credentials\linebreak%
    provided in the Connect service invocation allows for creating the named federation execution.
  }
  \HLAFixedBlock{90.000}{316.805}{433.120}{\rmfamily}{9.96}{12.00}{%
    The RTI shall load the supplied MIM if specified; otherwise, it shall automatically load the standard MIM\linebreak%
    (see Annex G). The MIM shall be loaded before any supplied FOM module is loaded. If a MIM argument is\linebreak%
    supplied, the supplied MIM module designator shall not be “HLAstandardMIM.”
  }
  \HLAFixedBlock{90.000}{362.764}{177.755}{\rmfamily\raggedright}{9.96}{12.00}{%
    At least one FOM module shall be supplied.
  }
  \HLAFixedBlock{90.000}{383.764}{433.219}{\rmfamily}{9.96}{12.00}{%
    The FOM modules supplied, together with the default or supplied MIM, shall result in a compliant FDD.\linebreak%
    The FOM module designators argument identifies the FOM modules that, together with the MIM, shall\linebreak%
    furnish the FDD for the federation execution to be created.
  }
