  \HLAFixedBlock{90.000}{635.826}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    The \emph{Object Instance Name Reserved \textdagger{}} service notifies the joined federate whether the name provided in a%
  }
  \HLAFixedBlock{90.000}{647.825}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    previous invocation of \emph{Reserve Object Instance Name} service has been reserved. A reservation-success%
  }
  \HLAFixedBlock{90.000}{659.825}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    indicator argument indicating success shall mean that the joined federate has obtained the unique right to%
  }
  \HLAFixedBlock{90.000}{671.825}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    register object instances with the supplied name. The reservation is retained until the joined federate either%
  }
  \HLAFixedBlock{90.000}{683.825}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    resigns or relinquishes the reservation by successfully invoking the \emph{Release Object Instance Name} service.%
  }
  \HLAFixedBlock{90.000}{695.825}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    While a joined federate holds a reservation on a name, no other federate joined to the current federation%
  }
  \HLAFixedBlock{90.000}{707.824}{432.000}{\rmfamily\raggedright}{9.96}{12.00}{%
    execution shall receive a successful reservation for that name. %
  }
