  \HLADefinitionAt{69.79}{%
    \HLADefinitionAnchor{time-axis}\HLADefinitionTerm{time axis}: A totally ordered sequence of discrete values in which each value typically represents an HLA\linebreak
    instant of time in the physical system being modeled. For any two points, T1 and T2, on the time axis, if T1\linebreak
    < T2, T1 represents an instant of time that occurs before the instant represented by T2. The sequence begins\linebreak
    with an initial value, for which no point is less, and ends with a final value, for which no point is greater.%
  }

  \HLADefinitionAt{123.78}{%
    \HLADefinitionAnchor{time-constrained-federate}\HLADefinitionTerm{time-constrained federate}: A joined federate that may receive timestamp order (TSO) messages and whose\linebreak
    time advances are constrained by other joined federates within a federation execution.%
  }
  \HLADefinitionNoteAt{90}{155.96}{NOTE---See 8.1.}

  \HLADefinitionAt{175.81}{%
    \HLADefinitionAnchor{time-management}\HLADefinitionTerm{time management}: A collection of High Level Architecture (HLA) services that support controlled\linebreak
    message ordering and delivery to the cooperating joined federates within a federation execution in a way\linebreak
    that is consistent with federation requirements.%
  }

  \HLADefinitionAt{218.77}{%
    \HLADefinitionAnchor{time-position}\HLADefinitionTerm{time position}: A joined federate's association with a point on the time axis.%
  }

  \HLADefinitionAt{240.78}{%
    \HLADefinitionAnchor{time-regulating-federate}\HLADefinitionTerm{time-regulating federate}: A joined federate that may send timestamp order (TSO) messages and that\linebreak
    constrains the time advances of other joined federates within a federation execution.%
  }
  \HLADefinitionNoteAt{90}{275.00}{NOTE---See 8.1.}

  \HLADefinitionAt{294.79}{%
    \HLADefinitionAnchor{timestamp-of-message-or-save}\HLADefinitionTerm{timestamp (of message or save)}: The association of a logical time with an activity of a federate (e.g.,\linebreak
    updating instance attribute values, sending interactions, and federation save).%
  }

  \HLADefinitionAt{328.81}{%
    \HLADefinitionAnchor{timestamp-order}\HLADefinitionTerm{timestamp order (TSO)}: An ordering of messages provided by a runtime infrastructure (RTI) for joined\linebreak
    federates making use of time management services and messages containing timestamps. Messages having\linebreak
    different timestamps are said to be delivered in TSO if for any two messages, M1 and M2 (timestamped with\linebreak
    T1 and T2, respectively), that are delivered to a single joined federate where T1 < T2, then M1 is delivered\linebreak
    before M2. Messages having the same timestamp will be delivered in an arbitrary order (i.e., no tie-breaking\linebreak
    mechanism is provided by an RTI).%
  }

  \HLADefinitionAt{410.76}{%
    \HLADefinitionAnchor{transportation-type}\HLADefinitionTerm{transportation type}: A runtime infrastructure (RTI) provided means of transmitting messages between\linebreak
    joined federates. Different categories of service are defined with different characteristics such as reliability\linebreak
    of delivery and message latency.%
  }

  \HLADefinitionAt{453.78}{%
    \HLADefinitionAnchor{unspecified-dimensions}\HLADefinitionTerm{unspecified dimensions}: The available dimensions of a class attribute, instance attribute, interaction class,\linebreak
    or sent interaction less the specified dimensions of the region specification from which the region realization\linebreak
    is derived.%
  }
